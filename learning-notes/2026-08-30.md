# 1. Confirm which branch you are using
git branch --show-current

# 2. See which files changed
git status

# 3. Review your changes
git diff

# 4. Stage all changes
git add .

# 5. Review exactly what will be committed
git diff --staged

# 6. Create the commit
git commit -m "Describe what you changed and learned"

# 7. Push it to the existing pull request
git push
